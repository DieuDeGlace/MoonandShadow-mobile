cmake_minimum_required(VERSION 3.20)
project(MoonandShadow LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 is required")
endif()
if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang" OR NOT MSVC)
  message(FATAL_ERROR "Use clang-cl in an x64 Visual Studio tools environment")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
function(configure_mod target)
  target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/vendor/cwmods")
  target_link_libraries(${target} PRIVATE d3d11 dxgi d3dcompiler user32 winmm)
endfunction()
add_library(MoonandShadow SHARED MoonandShadow.cpp)
configure_mod(MoonandShadow)
include(CTest)
option(ENABLE_GPU_TESTS "Enable D3D11 graphics tests" OFF)
if(BUILD_TESTING)
 add_executable(MoonRulesTest MoonLight/RulesTest.cpp)
 configure_mod(MoonRulesTest)
 target_compile_options(MoonRulesTest PRIVATE /UNDEBUG)
 add_test(NAME MoonRules COMMAND MoonRulesTest)
 if(ENABLE_GPU_TESTS)
  add_executable(MoonGraphicsTest MoonLight/GraphicsTest.cpp)
  configure_mod(MoonGraphicsTest)
  target_compile_options(MoonGraphicsTest PRIVATE /UNDEBUG)
  add_test(NAME MoonGraphicsTest COMMAND MoonGraphicsTest)
  add_executable(ShadowResourcesTest SteamShadows/SmokeTest.cpp)
  configure_mod(ShadowResourcesTest)
  target_compile_options(ShadowResourcesTest PRIVATE /UNDEBUG)
  add_test(NAME ShadowResourcesTest COMMAND ShadowResourcesTest)
  add_executable(FilteringTest SteamShadows/FilteringTest.cpp)
  configure_mod(FilteringTest)
  target_compile_options(FilteringTest PRIVATE /UNDEBUG)
  add_test(NAME FilteringTest COMMAND FilteringTest)
  add_executable(CloudShadowsTest SteamShadows/CloudShadowTest.cpp)
  configure_mod(CloudShadowsTest)
  target_compile_options(CloudShadowsTest PRIVATE /UNDEBUG)
  add_test(NAME CloudShadowsTest COMMAND CloudShadowsTest)
 endif()
endif()
